cmake_minimum_required(VERSION 3.10)
project(occupancy_avoidance)
set(CMAKE_CXX_STANDARD 14)
add_compile_options(-O3 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS message_generation geometry_msgs roscpp nav_msgs sensor_msgs std_msgs visualization_msgs waypoint_maker grid_map_core grid_map_ros tf2 morai_msgs)
find_package(octomap REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(JSONCPP REQUIRED jsoncpp)
find_package(Eigen3 REQUIRED)
add_message_files(FILES TunnelScene.msg TunnelObject.msg M8ApprovedPath.msg)
generate_messages(DEPENDENCIES std_msgs geometry_msgs nav_msgs)
catkin_package(INCLUDE_DIRS include LIBRARIES occupancy_ros_adapter occupancy_core CATKIN_DEPENDS message_runtime geometry_msgs roscpp nav_msgs sensor_msgs std_msgs visualization_msgs waypoint_maker grid_map_core grid_map_ros tf2 morai_msgs DEPENDS Eigen3 octomap)
set(VENDOR_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/vendor)
set(ENABLE_MKL_PARDISO OFF CACHE BOOL "" FORCE)
set(UNITTESTS OFF CACHE BOOL "" FORCE)
set(CTRLC OFF CACHE BOOL "" FORCE)
add_subdirectory(${VENDOR_ROOT}/occupancy_osqp ${CMAKE_CURRENT_BINARY_DIR}/upstream_osqp EXCLUDE_FROM_ALL)
set(APOLLO ${VENDOR_ROOT}/occupancy_apollo)
add_library(occupancy_core src/core.cpp src/joint_escape.cpp
  ${APOLLO}/modules/planning/planning_open_space/coarse_trajectory_generator/reeds_shepp_path.cc
  ${APOLLO}/modules/planning/planning_base/math/piecewise_jerk/piecewise_jerk_problem.cc
  ${APOLLO}/modules/planning/planning_base/math/piecewise_jerk/piecewise_jerk_path_problem.cc)
target_include_directories(occupancy_core PUBLIC include ${catkin_INCLUDE_DIRS} ${OCTOMAP_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS}
  PRIVATE adapters ${APOLLO})
target_compile_options(occupancy_core PRIVATE -fopenmp)
target_compile_options(occupancy_core PRIVATE -include array)
target_link_libraries(occupancy_core ${grid_map_core_LIBRARIES} ${OCTOMAP_LIBRARIES} osqpstatic gomp)
add_library(occupancy_ros_adapter src/ros_adapter.cpp)
target_include_directories(occupancy_ros_adapter PRIVATE ${JSONCPP_INCLUDE_DIRS})
target_link_libraries(occupancy_ros_adapter occupancy_core ${catkin_LIBRARIES} ${JSONCPP_LIBRARIES})
add_dependencies(occupancy_ros_adapter ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
add_executable(occupancy_avoidance_node src/node.cpp)
target_include_directories(occupancy_avoidance_node PRIVATE ${JSONCPP_INCLUDE_DIRS})
target_link_libraries(occupancy_avoidance_node occupancy_ros_adapter occupancy_core ${catkin_LIBRARIES} ${JSONCPP_LIBRARIES})
add_dependencies(occupancy_avoidance_node ${catkin_EXPORTED_TARGETS})
add_executable(tunnel_perception_scene_node src/scene_node.cpp)
target_link_libraries(tunnel_perception_scene_node occupancy_ros_adapter ${catkin_LIBRARIES})
add_dependencies(tunnel_perception_scene_node ${catkin_EXPORTED_TARGETS})
add_executable(occupancy_offline src/offline.cpp)
target_link_libraries(occupancy_offline occupancy_core ${JSONCPP_LIBRARIES})
target_include_directories(occupancy_offline PRIVATE ${JSONCPP_INCLUDE_DIRS})
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(occupancy_core_test test/core_test.cpp test/joint_escape_test.cpp)
  if(TARGET occupancy_core_test)
    target_link_libraries(occupancy_core_test occupancy_core)
  endif()
endif()
catkin_install_python(PROGRAMS scripts/export_bag.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(TARGETS occupancy_ros_adapter occupancy_core occupancy_avoidance_node occupancy_offline tunnel_perception_scene_node
  ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(FILES README.md THIRD_PARTY.md DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})

install(DIRECTORY vendor/occupancy_apollo/ DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/vendor/occupancy_apollo)
